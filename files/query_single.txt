    DWH_BUSINESS_DATE >= CAST('##START_DATE##' AS INT)
    AND UPPER(INSTANCE_NAME) IN ('RCNIPRD', 'RCNIUAT')
    AND RECON_ID IN (9774339, 9777858, 9776729, 9771338, 9777334)
    AND UPPER(BREAK_STATUS) IN ('OPEN', 'RESOLVED')
